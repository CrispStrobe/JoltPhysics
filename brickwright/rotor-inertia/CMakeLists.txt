cmake_minimum_required(VERSION 3.20)
project(RelativeRotorProbe LANGUAGES CXX)
set(BW_PHYSICS_ROOT "" CACHE PATH "Exact previous native source for the zero-inertia control")
option(BW_ROTOR_BASELINE "Compile against the source without the rotor API" OFF)
if(BW_PHYSICS_ROOT)
    set(PHYSICS_REPO_ROOT "${BW_PHYSICS_ROOT}")
else()
    get_filename_component(PHYSICS_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(OBJECT_LAYER_BITS 32)
set(INTERPROCEDURAL_OPTIMIZATION OFF)
include("${PHYSICS_REPO_ROOT}/Jolt/Jolt.cmake")
add_executable(rotor-inertia-probe probe.cpp)
target_link_libraries(rotor-inertia-probe PRIVATE Jolt)
set(BW_NATIVE_SOURCE_SHA "UNBOUND" CACHE STRING "Metadata-admitted engine source SHA")
set(BW_PROBE_SOURCE_SHA "UNBOUND" CACHE STRING "Exact probe source SHA")
target_compile_definitions(rotor-inertia-probe PRIVATE BW_NATIVE_SOURCE_SHA="${BW_NATIVE_SOURCE_SHA}" BW_PROBE_SOURCE_SHA="${BW_PROBE_SOURCE_SHA}")
if(BW_ROTOR_BASELINE)
    target_compile_definitions(rotor-inertia-probe PRIVATE BW_ROTOR_BASELINE)
else()
    add_executable(rotor-hinge-unit-tests
        "${PHYSICS_REPO_ROOT}/UnitTests/UnitTestFramework.cpp"
        "${PHYSICS_REPO_ROOT}/UnitTests/PhysicsTestContext.cpp"
        "${PHYSICS_REPO_ROOT}/UnitTests/Physics/HingeConstraintTests.cpp")
    target_include_directories(rotor-hinge-unit-tests PRIVATE "${PHYSICS_REPO_ROOT}/UnitTests")
    target_link_libraries(rotor-hinge-unit-tests PRIVATE Jolt)
endif()
