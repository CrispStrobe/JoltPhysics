cmake_minimum_required(VERSION 3.20)
project(RopeImpulseProbe LANGUAGES CXX)
set(BW_PHYSICS_ROOT "" CACHE PATH "Alternate producer root for the previous-version control")
if(BW_PHYSICS_ROOT)
    set(PHYSICS_REPO_ROOT "${BW_PHYSICS_ROOT}")
else()
    get_filename_component(PHYSICS_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
endif()
option(BW_DISTANCE_IMPULSE_BASELINE "Use the unchanged legacy getter control" OFF)
set(OBJECT_LAYER_BITS 32)
set(INTERPROCEDURAL_OPTIMIZATION OFF)
include("${PHYSICS_REPO_ROOT}/Jolt/Jolt.cmake")
add_executable(rope-impulse-probe distance-impulse-probe.cpp)
target_link_libraries(rope-impulse-probe PRIVATE Jolt)

if(BW_DISTANCE_IMPULSE_BASELINE)
    target_compile_definitions(rope-impulse-probe PRIVATE BW_DISTANCE_IMPULSE_BASELINE)
endif()
